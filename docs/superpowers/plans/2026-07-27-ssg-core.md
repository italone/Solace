# SSG Core Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a minimal `@italone/solace/server` static-site generation core that turns explicit routes into in-memory HTML pages on top of `renderToString()`.

**Architecture:** Keep SSG as a thin server-subpath layer, not a CLI or file-system pipeline. The new generator owns route validation, per-route rendering, optional shell wrapping, and result ordering; `renderToString()` stays the single renderer primitive. Public-package checks and packed-consumer smoke stay the guardrails for the exported surface.

**Tech Stack:** TypeScript, Vitest, pnpm, existing Solace server renderer and package export checks.

```ts
import type { Provides } from "../component/provide";
import type { RenderToStringSource } from "./render-to-string";
import { renderToString } from "./render-to-string";

export interface StaticRoute {
  path: string;
  source: RenderToStringSource;
  context?: Record<string, unknown>;
  provides?: Provides;
}

export interface StaticPage {
  path: string;
  html: string;
  body: string;
  styles: string[];
}

export interface GenerateStaticSiteOptions {
  routes: StaticRoute[];
  shell?: (page: {
    path: string;
    body: string;
    styles: string[];
    context: Record<string, unknown>;
  }) => string;
}

export interface GenerateStaticSiteResult {
  pages: StaticPage[];
}

export function generateStaticSite(options: GenerateStaticSiteOptions): GenerateStaticSiteResult {
  assertRoutes(options.routes);

  const seen = new Set<string>();
  const pages = options.routes.map((route) => {
    if (!route.path.startsWith("/")) {
      throw new TypeError(`SSG route path must start with "/": ${route.path}`);
    }

    if (seen.has(route.path)) {
      throw new TypeError(`Duplicate SSG route path: ${route.path}`);
    }

    seen.add(route.path);

    const context = route.context ?? {};
    const rendered = renderToString(route.source, {
      context,
      provides: route.provides,
    });
    const body = rendered.html;
    const html = options.shell
      ? options.shell({
          path: route.path,
          body,
          styles: rendered.styles,
          context,
        })
      : body;

    return {
      path: route.path,
      html,
      body,
      styles: rendered.styles,
    };
  });

  return { pages };
}

function assertRoutes(routes: StaticRoute[]): void {
  if (!Array.isArray(routes) || routes.length === 0) {
    throw new TypeError("SSG routes must be a non-empty array");
  }
}
```
